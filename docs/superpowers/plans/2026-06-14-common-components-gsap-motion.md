# Common Components GSAP Motion Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a targeted GSAP motion layer to the Vue common component library without changing backend behavior or broad page layout behavior.

**Architecture:** Add one shared `useGsapMotion` composable that owns motion presets, reduced-motion detection, and Vue Transition hook helpers. Update high-reuse common components to call that module instead of scattering raw GSAP calls. Keep `DataTable` desktop virtual rows untouched; only animate lightweight mobile cards and empty-state containers.

**Tech Stack:** Vue 3 Composition API, TypeScript, TailwindCSS, GSAP core, Pinia, Vitest with jsdom and Vue Test Utils.

---

## File Structure

- Create `frontend/src/composables/useGsapMotion.ts`: GSAP core import, reduced-motion detection, shared helpers, Vue Transition JS hooks, and mount/hover helpers.
- Create `frontend/src/composables/__tests__/useGsapMotion.spec.ts`: unit coverage for reduced-motion detection and hook behavior with mocked GSAP.
- Modify `frontend/package.json`: add `gsap` to dependencies.
- Modify `frontend/pnpm-lock.yaml`: generated by `pnpm --dir frontend install --lockfile-only`.
- Modify `frontend/src/components/common/BaseDialog.vue`: replace CSS-only transition with JS hooks from `useGsapMotion`.
- Modify `frontend/src/components/common/Toast.vue`: replace `TransitionGroup` CSS classes with JS hooks from `useGsapMotion`.
- Modify `frontend/src/components/common/Select.vue`: replace dropdown CSS transition with JS hooks from `useGsapMotion`; keep teleported positioning logic unchanged.
- Modify `frontend/src/components/common/EmptyState.vue`: add root/content refs and mount animation.
- Modify `frontend/src/components/common/StatCard.vue`: add root/content refs, mount animation, and transform-only hover hooks.
- Modify `frontend/src/components/common/DataTable.vue`: add refs and mount animation only for mobile cards and empty states; do not animate desktop virtual rows or table measurement logic.
- Modify `frontend/src/style.css`: remove obsolete `.modal-*` transition rules and add `will-change` only for elements that actually animate.
- Add or modify common component tests under `frontend/src/components/common/__tests__/`.

## Implementation Rules

- Use GSAP transform aliases (`x`, `y`, `scale`) and `autoAlpha`; do not animate layout properties.
- Use JS hooks on Vue `<Transition>` / `<TransitionGroup>` with `:css="false"` where GSAP owns enter/leave.
- Every hook must call `done` even in reduced-motion mode.
- Use `overwrite: 'auto'` or `overwrite: true` for interactive hover to avoid stacking tweens.
- Kill or complete relevant tweens during cleanup via the shared helper when a component unmounts.
- Do not add GSAP plugins.
- Do not touch existing Docker/deploy workspace changes.

```typescript
import { gsap } from 'gsap'

type DoneCallback = () => void

export const motionDurations = {
  fast: 0.16,
  normal: 0.26,
  expressive: 0.34
} as const

export const motionEases = {
  enter: 'power3.out',
  expressiveEnter: 'back.out(1.35)',
  exit: 'power2.in',
  hover: 'power2.out'
} as const

export function prefersReducedMotion(): boolean {
  if (typeof window === 'undefined' || typeof window.matchMedia !== 'function') {
    return false
  }
  return window.matchMedia('(prefers-reduced-motion: reduce)').matches
}

function completeImmediately(done: DoneCallback, target: Element, vars: gsap.TweenVars = {}) {
  gsap.set(target, vars)
  done()
}

function findPanel(el: Element): Element {
  return el.querySelector('.modal-content') ?? el
}

export function createDialogTransitionHooks() {
  return {
    onEnter(el: Element, done: DoneCallback) {
      const panel = findPanel(el)
      if (prefersReducedMotion()) {
        gsap.set(el, { autoAlpha: 1 })
        gsap.set(panel, { autoAlpha: 1, y: 0, scale: 1 })
        done()
        return
      }

      gsap.timeline({ onComplete: done })
        .fromTo(el, { autoAlpha: 0 }, { autoAlpha: 1, duration: motionDurations.fast, ease: motionEases.enter })
        .fromTo(
          panel,
          { autoAlpha: 0, y: 18, scale: 0.96 },
          { autoAlpha: 1, y: 0, scale: 1, duration: motionDurations.expressive, ease: motionEases.expressiveEnter },
          '<'
        )
    },
    onLeave(el: Element, done: DoneCallback) {
      const panel = findPanel(el)
      if (prefersReducedMotion()) {
        completeImmediately(done, el, { autoAlpha: 0 })
        return
      }

      gsap.timeline({ onComplete: done })
        .to(panel, { autoAlpha: 0, y: 10, scale: 0.98, duration: motionDurations.fast, ease: motionEases.exit })
        .to(el, { autoAlpha: 0, duration: motionDurations.fast, ease: motionEases.exit }, '<')
    }
  }
}

export function createToastTransitionHooks() {
  return {
    onEnter(el: Element, done: DoneCallback) {
      if (prefersReducedMotion()) {
        completeImmediately(done, el, { autoAlpha: 1, x: 0, scale: 1 })
        return
      }

      gsap.fromTo(
        el,
        { autoAlpha: 0, x: 44, scale: 0.96 },
        { autoAlpha: 1, x: 0, scale: 1, duration: motionDurations.normal, ease: motionEases.expressiveEnter, onComplete: done }
      )
    },
    onLeave(el: Element, done: DoneCallback) {
      if (prefersReducedMotion()) {
        completeImmediately(done, el, { autoAlpha: 0 })
        return
      }

      gsap.to(el, { autoAlpha: 0, x: 36, scale: 0.98, duration: motionDurations.fast, ease: motionEases.exit, onComplete: done })
    },
    onMove(el: Element, done: DoneCallback) {
      if (prefersReducedMotion()) {
        done()
        return
      }

      gsap.fromTo(el, { scale: 0.98 }, { scale: 1, duration: motionDurations.fast, ease: motionEases.enter, onComplete: done })
    }
  }
}

export function createSelectTransitionHooks() {
  return {
    onEnter(el: Element, done: DoneCallback) {
      if (prefersReducedMotion()) {
        completeImmediately(done, el, { autoAlpha: 1, y: 0, scaleY: 1 })
        return
      }

      const options = Array.from(el.querySelectorAll('.select-option'))
      const timeline = gsap.timeline({ onComplete: done })
      timeline.fromTo(
        el,
        { autoAlpha: 0, y: -6, scaleY: 0.96, transformOrigin: 'top center' },
        { autoAlpha: 1, y: 0, scaleY: 1, duration: motionDurations.normal, ease: motionEases.expressiveEnter }
      )
      if (options.length > 0) {
        timeline.fromTo(
          options.slice(0, 8),
          { autoAlpha: 0, y: 4 },
          { autoAlpha: 1, y: 0, duration: 0.18, stagger: 0.018, ease: motionEases.enter },
          '-=0.18'
        )
      }
    },
    onLeave(el: Element, done: DoneCallback) {
      if (prefersReducedMotion()) {
        completeImmediately(done, el, { autoAlpha: 0 })
        return
      }

      gsap.to(el, { autoAlpha: 0, y: -4, scaleY: 0.98, duration: motionDurations.fast, ease: motionEases.exit, onComplete: done })
    }
  }
}

export function animateMountedSurface(el: Element | null, childSelector?: string) {
  if (!el) return
  if (prefersReducedMotion()) {
    gsap.set(el, { autoAlpha: 1, y: 0, scale: 1 })
    return
  }

  const children = childSelector ? Array.from(el.querySelectorAll(childSelector)) : []
  const timeline = gsap.timeline()
  timeline.fromTo(
    el,
    { autoAlpha: 0, y: 14, scale: 0.985 },
    { autoAlpha: 1, y: 0, scale: 1, duration: motionDurations.normal, ease: motionEases.enter }
  )
  if (children.length > 0) {
    timeline.fromTo(
      children,
      { autoAlpha: 0, y: 6 },
      { autoAlpha: 1, y: 0, duration: 0.2, stagger: 0.04, ease: motionEases.enter },
      '-=0.16'
    )
  }
}

export function animateHoverLift(el: Element | null, lifted: boolean) {
  if (!el || prefersReducedMotion()) return
  gsap.killTweensOf(el)
  gsap.to(el, {
    y: lifted ? -3 : 0,
    scale: lifted ? 1.01 : 1,
    duration: motionDurations.fast,
    ease: motionEases.hover,
    overwrite: true
  })
}

export function clearMotion(el: Element | Element[] | null) {
  if (!el) return
  gsap.killTweensOf(el)
}
```
